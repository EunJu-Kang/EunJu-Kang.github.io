## 프로그래머스 [완주하지 못한 선수] 
[문제보기](https://school.programmers.co.kr/learn/courses/30/lessons/42576) <br>

### 소스코드

~~~ java
import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;

class Solution {
    public String solution(String[] participant, String[] completion) {
        Arrays.sort(participant);
        Arrays.sort(completion);

        int paticipantSize = participant.length;
        int completionSize = paticipantSize - 1;

        for (int i = 0; i < paticipantSize; i++) {
            String participantName = participant[i];
            if (completionSize <=i || !participantName.equals(completion[i])) return participantName;

        }

        return "";
    }
}
~~~
